# NFL Conferences & Divisions Mapping NFL_CONFERENCES = { "1": "AFC", "2": "NFC", # NFL Divisions (if using division group IDs) "7": "AFC East", "8": "AFC North", "9": "AFC South", "10": "AFC West", "11": "NFC East", "12": "NFC North", "13": "NFC South", "14": "NFC West", } # College Football (FBS) Major Group IDs CFB_CONFERENCES = { "1": "ACC", "4": "Big 12", "8": "SEC", "9": "Big Ten", "12": "Pac-12", "15": "Conference USA", "17": "MAC", "18": "MWC", "37": "Sun Belt", "151": "American", "80": "FBS Independents", } def get_conference_name(group_id, sport="nfl"): group_str = str(group_id) if sport == "nfl": return NFL_CONFERENCES.get(group_str, f"Conference {group_str}") elif sport in ["cfb", "college-football"]: return CFB_CONFERENCES.get(group_str, f"Conference {group_str}") return f"Group {group_str}" # Usage Example raw_group_id = 1 print( get_conference_name(raw_group_id, sport="nfl") ) # Output: AFC
